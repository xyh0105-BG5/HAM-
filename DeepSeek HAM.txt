<div class="chat-container">
    <div class="header">📻 HAM-AI (DeepSeek 驱动版)</div>
    <div class="chat-box" id="chatBox">
        <div class="message bot">你好！我是接入真实大模型的 HAM 助手。你可以问我任何业余无线电、天线制作、刷机避坑的问题，甚至可以让我帮你计算驻波或设计天线。</div>
    </div>
    <div class="input-area">
        <input type="text" id="userInput" placeholder="输入你的问题..." onkeypress="handleKeyPress(event)">
        <button id="sendBtn" onclick="sendMessage()">发送</button>
    </div>

<script>
    // ⚠️ 把这里替换成复制的 API KEY，去DeepSeek官网或dsh里搞
    const API_KEY = "sk-你的API key填这里"; 
    
    // DeepSeek 的对话接口
    const API_URL = "https://api.deepseek.com/chat/completions";

    // 记忆功能：保存对话历史，让 AI 知道你们聊过什么
    let chatHistory = [
        { role: "system", content: "你是一个极其专业的资深业余无线电火腿（HAM），熟悉UV-K6刷机、天线制作、无线电法规、短波通信等。回答要幽默、硬核、接地气，多用火腿圈的黑话。" }
    ];

    function handleKeyPress(e) {
        if (e.key === 'Enter') { sendMessage(); }
    }

    async function sendMessage() {
        const input = document.getElementById('userInput');
        const sendBtn = document.getElementById('sendBtn');
        const text = input.value.trim();
        if (!text) return;

        if (API_KEY.includes("填在这里")) {
            alert("请先打开代码，把你的 API Key 填进去！");
            return;
        }

        // 1. 界面显示用户消息
        appendMessage(text, 'user');
        input.value = '';
        
        // 2. 锁定按钮，显示加载中
        sendBtn.disabled = true;
        sendBtn.innerText = "思考中";
        const loadingId = appendMessage("正在调用 DeepSeek 大脑...", 'bot');

        // 3. 更新对话历史
        chatHistory.push({ role: "user", content: text });

        try {
            // 4. 发送请求给 API
            const response = await fetch(API_URL, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${API_KEY}`
                },
                body: JSON.stringify({
                    model: "deepseek-chat",
                    messages: chatHistory,
                    temperature: 0.7
                })
            });

            const data = await response.json();
            
            // 5. 移除加载提示，显示 AI 的真实回答
            document.getElementById(loadingId).remove();
            
            if (data.choices && data.choices.length > 0) {
                const botReply = data.choices[0].message.content;
                appendMessage(botReply, 'bot');
                chatHistory.push({ role: "assistant", content: botReply });
            } else {
                appendMessage("API 报错了，可能是余额不足或 Key 填错了。", 'bot');
            }

        } catch (error) {
            document.getElementById(loadingId).remove();
            appendMessage("网络请求失败！请检查你的网络，或者是不是跨域被拦截了。", 'bot');
            console.error(error);
        } finally {
            // 6. 恢复按钮
            sendBtn.disabled = false;
            sendBtn.innerText = "发送";
        }
    }

    function appendMessage(text, sender) {
        const chatBox = document.getElementById('chatBox');
        const msgDiv = document.createElement('div');
        msgDiv.className = `message ${sender}`;
        msgDiv.innerText = text;
        const id = "msg-" + Date.now();
        msgDiv.id = id;
        chatBox.appendChild(msgDiv);
        chatBox.scrollTop = chatBox.scrollHeight;
        return id;
    }
</script>
